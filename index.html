backend/controllers/dispatchController.js 
async function autoAssignProvider(serviceType, orderId) {
  // Query certified technicians matching category and capacity
  const candidates = await Provider.find({
    specialty: serviceType,
    status: 'available',
    activeJobsCount: { $lt: 3 }
  }).lean();

  if (!candidates.length) throw new Error('No available technicians');

  // Fair uniform random selection among top eligible providers
  const chosen = candidates[Math.floor(Math.random() * candidates.length)];

  // Atomic state change: 'requested' -> 'assigned' in <100ms
  await Order.findByIdAndUpdate(orderId, {
    status: 'assigned',
    providerId: chosen._id,
    assignedAt: new Date()
  });

  io.to(chosen._id.toString()).emit('NEW_JOB_ALERT', { orderId });
  return chosen;
}
